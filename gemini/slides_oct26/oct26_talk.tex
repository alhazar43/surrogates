  \textbf{\color{NavyHeader}\small Central Question}\\[1mm]
  Can static item parameters $(a_j, b_j)$ calibrated across leaderboard LLMs guide adaptive evaluation and reinforcement learning—and when they fail, what can item-level rollout distributions tell us instead?\par

  \vspace{2.5mm}
  {\color{BorderGray}\rule{\textwidth}{0.5pt}}
  \vspace{2mm}

  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.31\textwidth}
      \textbf{\color{RedGRPO}1. The Evaluation \& RL Gap}\\[1mm]
      Outcome RL (\texttt{GRPO}) beats \texttt{SFT} on single-draw $\mathrm{Pass@1}$, yet collapses on $\mathrm{Pass@16}$ and adds \textbf{zero net gain} on Majority Vote ($\mathrm{Maj@16}$). Plugging leaderboard 2PL into adaptive testing (\texttt{CAT}) or RL critics hits a structural ceiling.\par
    \end{column}
    \begin{column}{0.32\textwidth}
      \textbf{\color{BlueSFT}2. Rollout Diagnostics}\\[1mm]
      Without retraining, multi-draw rollouts separate \textbf{policy sharpening} ($\tau \leftrightarrow 1/T$) from \textbf{distractor overthrow}, exposing why RL locks in wrong modal beliefs while wasting over half of active updates in the $\le 25\%$ guessing region.\par
    \end{column}
    \begin{column}{0.31\textwidth}
      \textbf{\color{PurpleIRT}3. Dynamic Capability Tracking}\\[1mm]
      Static standing $\neq$ gradient plasticity ($r=0.072$). Guiding multi-step training requires a state-space formulation that separates epistemic filtering from answer-blind plasticity via an asymmetric teach–measure kernel.\par
    \end{column}
  \end{columns}
\end{frame}

        \[
          \hat{A}_j = 2\,a_j^\gamma\bigl(r_j - \hat{p}_j(\theta_t)\bigr), \qquad \gamma \in \{0, 1\}
        \]
      }
    \end{column}
  \end{columns}

% SLIDE 13: WHY STATIC SNAPSHOTS CANNOT GUIDE DYNAMIC TRAINING
% ==============================================================================
\begin{frame}{Why Static Snapshots Cannot Guide Dynamic Training}
  \footnotesize
  \vspace{-1.5mm}

  \noindent\textbf{\color{NavyHeader}1. Non-Ergodicity: Cross-Model Standing $\neq$ Within-Model Gradient Velocity}\\[1mm]
  {\centering $\displaystyle a_j^{\mathrm{dyn}}(t) \;\triangleq\; \frac{d\,\mathrm{logit}\,P_j(\boldsymbol{\phi}_t)}{dt} \;=\; \underbrace{\nabla_{\boldsymbol{\phi}}\mathrm{logit}\,P_j(\boldsymbol{\phi}_t)^\top}_{\text{Item Jacobian }\boldsymbol{J}_j}\;\underbrace{\dot{\boldsymbol{\phi}}_t}_{\text{Update Velocity }\boldsymbol{u}_t} \;\perp\; a_j^{\mathrm{pop}} \quad (r = 0.072)$\par}\vspace{1mm}
  Static rollout diagnostics ($\hat{p}_{j,0}, l_z, I_j^{\mathrm{3PL}}$) are single-checkpoint snapshots. Across training steps, logit evolution combines \textbf{global sharpening ($\tau_t$)}, \textbf{global drift ($\delta_t$)}, and \textbf{local support shift ($\Delta\kappa_{t,j}$)}.\par

  \vspace{1.5mm}
  \noindent{\color{BorderGray}\rule{\linewidth}{0.4pt}}\\[1.2mm]
  \textbf{\color{NavyHeader}2. The Second-Order $\mathcal{O}(\Delta^2)$ Sink Principle}\\[1mm]
  {\centering $\displaystyle \hat{\theta}_W - \theta_W^* \;\approx\; \mathcal{J}_{W\cdot o}^{-1}\textstyle\sum_{t=1}^T \mathbb{E}\bigl[p_t(1-p_t)\,\delta_t\,g_t\bigr]$\par}\vspace{1mm}
  Step-to-step skill transitions enter binary outcomes at \textbf{second order $\mathcal{O}(\Delta^2)$}, acting as a statistical sink for any unmodeled $\mathcal{O}(1)$ variation (temperature $\tau_t$, baseline standing, guessing floor). Identifying transitions requires \textbf{closing $\mathcal{O}(1)$ nuisances first} and evaluating over \textbf{multi-step open-loop horizons ($H > 1$)}.\par

  \vspace{1.5mm}
  \noindent{\color{BorderGray}\rule{\linewidth}{0.4pt}}\\[1.2mm]
  \textbf{\color{NavyHeader}3. Subspace Absorption \& Read--Write Asymmetry}\\[1mm]
  {\centering $\displaystyle \mathcal{P}_{\perp}(\boldsymbol{W}) \;=\; (\boldsymbol{I} - \boldsymbol{A}\boldsymbol{A}^+)\boldsymbol{W}, \qquad \boldsymbol{a}_j^{\mathrm{read}} \;\neq\; \boldsymbol{\psi}_j^{\mathrm{write}}$\par}\vspace{1mm}
  Unconstrained global trajectories absorb transitions inside $\mathrm{col}(\boldsymbol{A})$. Moreover, forcing a single item vector to serve as both \textbf{static read key ($\boldsymbol{a}_j^{\mathrm{read}}$)} and \textbf{dynamic write vector ($\boldsymbol{\psi}_j^{\mathrm{write}}$)} collapses multi-skill clusters into item memorization.\par
\end{frame}

\begin{frame}{A State-Space Model of Training Dynamics}
  \footnotesize
  \vspace{-1.5mm}

  \noindent\textbf{\color{NavyHeader}1. Hierarchical Observation Model \& Doob Martingale Decomposition}\\[1mm]
  {\centering $\displaystyle P_{j,t} = \sigma\!\Bigl(\tau_t\bigl(\boldsymbol{m}_j^\top \boldsymbol{v}_t + \boldsymbol{a}_j^\top \boldsymbol{z}_t + \boldsymbol{u}_j^\top \boldsymbol{c}_t - b_j\bigr)\Bigr), \qquad \Delta\boldsymbol{\kappa}_t = \underbrace{\boldsymbol{K}_t(y_t - \bar{p}_t)}_{\text{Epistemic Filter }\Delta\boldsymbol{z}_t\;(\mathbb{E}[\Delta\boldsymbol{z}_t]=\mathbf{0})} + \underbrace{\boldsymbol{g}_t(\bar{p}_t, \boldsymbol{\psi}_{e_t})}_{\text{Answer-Blind Drift }\Delta\boldsymbol{c}_t}$\par}\vspace{1mm}
  Frozen read keys $\boldsymbol{a}_j$ are identified from static residual covariance ($r=0.81$). Two structural zeros separate \textbf{epistemic filtering} ($\Delta\boldsymbol{z}_t$, zero drift on frozen checkpoints) from \textbf{answer-blind plasticity} ($\Delta\boldsymbol{c}_t$, which never reads $y_t$).\par

  \vspace{1.5mm}
  \noindent{\color{BorderGray}\rule{\linewidth}{0.4pt}}\\[1.2mm]
  \textbf{\color{NavyHeader}2. 3PL Interior Plasticity Law \& Preconditioned State Transition}\\[1mm]
  {\centering $\displaystyle \mathrm{Cov}_{\pi_\phi}\bigl(r,\,\nabla_\phi\log\pi_\phi\bigr) = p_j(1-p_j)\boldsymbol{\psi}_j \;\;\Longrightarrow\;\; \boldsymbol{c}_{t+1} = (\boldsymbol{I} - \boldsymbol{\Lambda})\,\boldsymbol{c}_t + \underbrace{c\,(\bar{p}_t - c_0)_+^{k_1}(1-\bar{p}_t)^{k_2}}_{\text{3PL Interior Gate }u(\bar{p}_t)}\,(1+\boldsymbol{s}_t)^{-\beta} \odot \boldsymbol{\psi}_{e_t}$\par}\vspace{1mm}
  Unifies SGD ($k=1$), std-normalized RL ($k=\tfrac{1}{2}$), and the 3PL guessing floor $c_0=0.25$ ($[2..6/8]$ rollout band), while $(1+\boldsymbol{s}_t)^{-\beta}$ ($\boldsymbol{s}_{t+1} = \boldsymbol{s}_t + \boldsymbol{g}_t^2$) captures second-moment saturation and $\boldsymbol{\Lambda} \in (0,1)^M$ controls consolidation timescales.\par

  \vspace{1.5mm}
  \noindent{\color{BorderGray}\rule{\linewidth}{0.4pt}}\\[1.2mm]
  \textbf{\color{NavyHeader}3. Asymmetric Teach--Measure Kernel \& Multi-Step Impulse Operator}\\[1mm]
  {\centering $\displaystyle K_{BA} \;=\; \boldsymbol{a}_B^\top \boldsymbol{\psi}_A \;\longleftrightarrow\; \boldsymbol{J}_{\mathrm{eval}}\,\boldsymbol{J}_{\mathrm{train}}^\top, \qquad W_\ell[j,i] \;=\; \boldsymbol{H}\boldsymbol{F}^{\ell-1}\boldsymbol{G}$\par}\vspace{1mm}
  Unties \textbf{Read Key $\boldsymbol{a}_B$} from \textbf{Write Vector $\boldsymbol{\psi}_A$}: diagonal $K_{AA}$ measures skill consolidation, off-diagonal $K_{BA}$ predicts cross-domain transfer vs.\ interference, and the Hankel spectrum of $W_\ell$ identifies the effective update rank.\par
\end{frame}
